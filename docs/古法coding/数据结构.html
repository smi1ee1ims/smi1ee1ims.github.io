    import sys
 
def dijkstra(graph, start_node):
    unseen = {node: sys.maxsize for node in graph}  # 初始化所有节点距离为无穷大
    unseen[start_node] = 0  # 起始节点距离为0
    shortest_paths = {start_node: (0, [])}  # 起始节点的路径和距离
 
    while unseen:
        current_node = min(unseen, key=unseen.get)  # 找到未访问节点中距离最小的节点
        current_distance = unseen[current_node]
 
	        for neighbor, distance in graph[current_node].items():
            if neighbor not in unseen: 
	            continue  # 已访问过的节点跳过
		        new_distance = current_distance + distance
            if new_distance < unvisited_nodes[neighbor]:  # 如果找到更短路径，更新
                unvisited_nodes[neighbor] = new_distance
                shortest_paths[neighbor] = (new_distance, shortest_paths[current_node][1] + [current_node])  # 更新路径和距离
 
        unvisited_nodes.pop(current_node)  # 当前节点已访问过，从未访问节点中删除
 
    return shortest_paths  # 返回最短路径和距离

#递归
def invertTree(self,root):
	cur=root
	if not cur:
		return
	left=cur.left #暂存左节点，用于左右交换的中间变量。也可以等号赋值交换
	cur.left=self.invertTree(cur.right)
	cur.right=self.inverTree(left)
	return cur

def binary_search_tree(arr):
    bst={}
    id=0
    bst[id]=[None,arr[0],None]
    
    for val in arr[1:]:
        id+=1
        current_id=0
        while True:
            current_val=bst[current_id][1]
            if val < current_val:
                if bst[current_id][0] is None:
                    bst[current_id][0]=id
                    bst[id]=[None,val,None]
                    break
                else:
                    current_id=bst[current_id][0]
            else:
                if bst[current_id][2] is None:
                    bst[current_id][2]=id
                    bst[id]=[None,val,None]
                    break
                else:
                    current_id=bst[current_id][2]
    return bst

    head=forwards[0]
    mids_left=[]
    for i in range(len(mids)):
        cur=mids[i]
        if cur != head:
            mids_left.append(cur)
        else:
            break
    mids_right=mids[i+1:]

    fwd_left=forwards[1:1+len_l]
    fwd_right=forwards[1+len_l:]
    if fwd_left and fwd_right:
